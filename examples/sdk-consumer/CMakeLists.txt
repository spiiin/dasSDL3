cmake_minimum_required(VERSION 3.24)
project(dasSDL3Consumer LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(dasSDL3 0.1 CONFIG REQUIRED)
add_executable(interpreter host.cpp)
target_link_libraries(interpreter PRIVATE dasSDL3::dasSDL3)
add_executable(aot host.cpp)
dassdl3_add_aot(aot SCRIPTS main.das MODULES dassdl3/sdl3_result dassdl3/sdl3_boost dassdl3/sdl3_audio_boost)
include(CTest)
add_test(NAME interpreter COMMAND interpreter "${CMAKE_CURRENT_SOURCE_DIR}/main.das" "${dasSDL3_DATA_DIR}")
add_test(NAME aot COMMAND aot "${CMAKE_CURRENT_SOURCE_DIR}/main.das" "${dasSDL3_DATA_DIR}" --aot)

add_test(NAME no_aot_fallback COMMAND interpreter "${CMAKE_CURRENT_SOURCE_DIR}/main.das" "${dasSDL3_DATA_DIR}" --aot)
set_tests_properties(no_aot_fallback PROPERTIES WILL_FAIL TRUE)
