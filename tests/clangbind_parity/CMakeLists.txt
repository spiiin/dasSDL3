cmake_minimum_required(VERSION 3.24)
project(SDLClangBindParity LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Parity currently validates Windows x64/MSVC only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DAS "${ROOT}/third_party/daScript")
set(MAIN_BUILD "${ROOT}/build/ninja" CACHE PATH "Existing SDL build")
set(SDL_INCLUDE "${MAIN_BUILD}/_deps/sdl3-src/include" CACHE PATH "SDL include")
set(LLVM_SDK "" CACHE PATH "LLVM 22.1 SDK")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(GLOB SDL_HEADERS CONFIGURE_DEPENDS "${SDL_INCLUDE}/SDL3/*.h")
add_custom_command(OUTPUT "${GEN}/parity_functions.inc" "${GEN}/parity-api.json" "${GEN}/parity_types.inc" "${GEN}/parity-contract.tsv"
    COMMAND ${Python3_EXECUTABLE} "${ROOT}/tools/run_clangbind_parity.py"
        --daslang "${DAS}/bin/daslang.exe" --sdl-include "${SDL_INCLUDE}" --sdk "${LLVM_SDK}" --output "${GEN}"
    DEPENDS "${ROOT}/tools/clangbind_parity.das" "${ROOT}/tools/run_clangbind_parity.py"
        "${ROOT}/tools/bindings.json" "${ROOT}/tools/api-policy.json" "${ROOT}/src/generated/api.json" ${SDL_HEADERS}
    VERBATIM)
add_custom_target(generate_parity DEPENDS "${GEN}/parity_functions.inc")
foreach(backend baseline cppgenbind)
    add_library(${backend}_module STATIC "${ROOT}/src/module_sdl3.cpp")
    target_include_directories(${backend}_module PUBLIC "${ROOT}/src" "${GEN}" "${DAS}/include" "${DAS}/3rdparty/fmt/include" "${SDL_INCLUDE}")
    target_compile_definitions(${backend}_module PUBLIC DAS_ENABLE_EXCEPTIONS=1 DAS_FUSION=2 DAS_NO_ASSERTIONS
        DASSDL3_TESTING=1 DASSDL3_DAS_ROOT="${DAS}" DASSDL3_MODULE_ROOT="${ROOT}/dassdl3")
    target_compile_options(${backend}_module PUBLIC /EHsc /bigobj /arch:AVX2)
    target_link_libraries(${backend}_module PUBLIC "${DAS}/lib/libDaScript.lib" "${DAS}/lib/libDaScript_runtime.lib"
        "${DAS}/lib/libUriParser.lib" "${MAIN_BUILD}/_deps/sdl3-build/SDL3-static.lib"
        dbghelp ws2_32 mswsock rpcrt4 kernel32 user32 gdi32 winmm imm32 ole32 oleaut32 version uuid advapi32 setupapi shell32 dinput8)
    add_executable(${backend}_runner "${ROOT}/src/runner.cpp")
    target_link_libraries(${backend}_runner PRIVATE ${backend}_module)
    add_executable(${backend}_metadata metadata.cpp)
    target_link_libraries(${backend}_metadata PRIVATE ${backend}_module)
endforeach()
add_dependencies(cppgenbind_module generate_parity)
target_compile_definitions(cppgenbind_module PRIVATE DASSDL3_REGISTRATION_INCLUDE="parity_functions.inc" DASSDL3_TYPES_INCLUDE="parity_types.inc")
target_compile_definitions(cppgenbind_module PRIVATE DASSDL3_AOT_HEADER="sdl3_parity_aot.h")
target_include_directories(cppgenbind_module PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${ROOT}/tests")
add_executable(parity_aot_tool aot_tool.cpp)
target_link_libraries(parity_aot_tool PRIVATE cppgenbind_module)
set(AOT_SCRIPTS tests/bindings tests/boost tests/textures tests/input tests/audio examples/02_square examples/04_textures examples/03_input examples/08_audio tests/clangbind_parity/recover)
list(APPEND AOT_SCRIPTS tests/pixels examples/05_streaming_texture examples/06_render_target)
list(APPEND AOT_SCRIPTS tests/geometry examples/07_geometry)
list(APPEND AOT_SCRIPTS tests/gpu_state tests/gpu examples/09_gpu_clear)
list(APPEND AOT_SCRIPTS tests/gpu_windows)
list(APPEND AOT_SCRIPTS tests/gpu_devices)
list(APPEND AOT_SCRIPTS tests/gpu_triangle examples/10_gpu_triangle)
list(APPEND AOT_SCRIPTS tests/gpu_mesh examples/11_gpu_textured_quad)
set(AOT_MODULES dassdl3/sdl3_boost dassdl3/sdl3_audio_boost dassdl3/sdl3_pixels_boost)
list(APPEND AOT_MODULES dassdl3/sdl3_geometry_boost)
list(APPEND AOT_MODULES dassdl3/sdl3_gpu_boost)
list(APPEND AOT_MODULES dassdl3/sdl3_gpu_mesh_boost)
foreach(script IN LISTS AOT_SCRIPTS AOT_MODULES)
    string(REPLACE "/" "_" name "${script}")
    set(output "${CMAKE_CURRENT_BINARY_DIR}/${name}.aot.cpp")
    add_custom_command(OUTPUT "${output}"
        COMMAND parity_aot_tool "${ROOT}/${script}.das" "${output}"
        DEPENDS parity_aot_tool "${ROOT}/${script}.das" "${ROOT}/dassdl3/sdl3_boost.das" "${ROOT}/dassdl3/sdl3_audio_boost.das" "${ROOT}/dassdl3/sdl3_pixels_boost.das" "${ROOT}/dassdl3/sdl3_geometry_boost.das" "${ROOT}/dassdl3/sdl3_gpu_boost.das" "${ROOT}/dassdl3/sdl3_gpu_mesh_boost.das"
        VERBATIM)
    list(APPEND AOT_OUTPUTS "${output}")
endforeach()
add_executable(parity_aot_runner aot_runner.cpp ${AOT_OUTPUTS})
target_link_libraries(parity_aot_runner PRIVATE cppgenbind_module)
add_executable(parity_missing_aot aot_runner.cpp)
target_link_libraries(parity_missing_aot PRIVATE cppgenbind_module)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/assets")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/assets/shaders")
foreach(shader triangle.vert.spv triangle.frag.spv triangle.vert.dxil triangle.frag.dxil triangle.vert.hlsl)
    configure_file("${ROOT}/examples/assets/shaders/${shader}" "${CMAKE_CURRENT_BINARY_DIR}/assets/shaders/${shader}" COPYONLY)
endforeach()
foreach(shader mesh.vert.spv mesh.frag.spv mesh.vert.dxil mesh.frag.dxil)
    configure_file("${ROOT}/examples/assets/shaders/${shader}" "${CMAKE_CURRENT_BINARY_DIR}/assets/shaders/${shader}" COPYONLY)
endforeach()
configure_file("${ROOT}/examples/assets/checker.bmp" "${CMAKE_CURRENT_BINARY_DIR}/assets/checker.bmp" COPYONLY)
configure_file("${ROOT}/examples/assets/tone.wav" "${CMAKE_CURRENT_BINARY_DIR}/assets/tone.wav" COPYONLY)
enable_testing()
foreach(driver vulkan direct3d12)
    add_test(NAME aot_gpu_devices_${driver} COMMAND parity_aot_runner "${ROOT}/tests/gpu_devices.das")
    set_tests_properties(aot_gpu_devices_${driver} PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 ENVIRONMENT "SDL_GPU_DRIVER=${driver}")
endforeach()
foreach(driver vulkan direct3d12)
    add_test(NAME aot_gpu_mesh_${driver} COMMAND parity_aot_runner "${ROOT}/tests/gpu_mesh.das")
    set_tests_properties(aot_gpu_mesh_${driver} PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 ENVIRONMENT "SDL_GPU_DRIVER=${driver}")
endforeach()
foreach(driver vulkan direct3d12)
    add_test(NAME aot_gpu_triangle_${driver} COMMAND parity_aot_runner "${ROOT}/tests/gpu_triangle.das")
    set_tests_properties(aot_gpu_triangle_${driver} PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 ENVIRONMENT "SDL_GPU_DRIVER=${driver}")
endforeach()
add_test(NAME sdl3_parity_missing_aot COMMAND ${Python3_EXECUTABLE}
    "${ROOT}/tests/clangbind/expect_aot_failure.py" $<TARGET_FILE:parity_missing_aot> "${ROOT}/tests/bindings.das")
foreach(script IN LISTS AOT_SCRIPTS)
    string(REPLACE "/" "_" name "${script}")
    add_test(NAME aot_${name} COMMAND parity_aot_runner "${ROOT}/${script}.das")
    set_tests_properties(aot_${name} PROPERTIES TIMEOUT 30)
    if(script MATCHES "(tests/gpu|tests/gpu_windows|tests/gpu_devices|tests/gpu_mesh|tests/gpu_triangle|examples/11_gpu_textured_quad|examples/10_gpu_triangle|examples/09_gpu_clear)$")
        set_tests_properties(aot_${name} PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
    endif()
    if(script MATCHES "audio$")
        set_tests_properties(aot_${name} PROPERTIES ENVIRONMENT "SDL_AUDIO_DRIVER=dummy")
    endif()
endforeach()
add_test(NAME sdl3_parity_generation_errors COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/test_generation_errors.py"
    "${ROOT}/tools/run_clangbind_parity.py" "${DAS}/bin/daslang.exe" "${LLVM_SDK}" "${SDL_INCLUDE}")
set_tests_properties(sdl3_parity_generation_errors PROPERTIES TIMEOUT 30)
add_test(NAME sdl3_parity_metadata COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/compare_metadata.py"
    $<TARGET_FILE:baseline_metadata> $<TARGET_FILE:cppgenbind_metadata>)
foreach(backend baseline cppgenbind)
    foreach(script IN LISTS AOT_SCRIPTS)
        string(REPLACE "/" "_" name "${script}")
        add_test(NAME ${backend}_${name} COMMAND ${backend}_runner "${ROOT}/${script}.das" --smoke-test)
        set_tests_properties(${backend}_${name} PROPERTIES TIMEOUT 30)
        if(script MATCHES "(tests/gpu|tests/gpu_windows|tests/gpu_devices|tests/gpu_mesh|tests/gpu_triangle|examples/11_gpu_textured_quad|examples/10_gpu_triangle|examples/09_gpu_clear)$")
            set_tests_properties(${backend}_${name} PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
        endif()
        if(script MATCHES "audio$")
            set_tests_properties(${backend}_${name} PROPERTIES ENVIRONMENT "SDL_AUDIO_DRIVER=dummy")
        endif()
    endforeach()
endforeach()
set(DASSDL3_TEST_VULKAN_LAYERS_DISABLE "" CACHE STRING "Optional Vulkan layer filter for GPU tests")
if(DASSDL3_TEST_VULKAN_LAYERS_DISABLE)
    get_property(_tests DIRECTORY PROPERTY TESTS)
    foreach(_test IN LISTS _tests)
        if(_test MATCHES "gpu")
            set_property(TEST ${_test} APPEND PROPERTY ENVIRONMENT "VK_LOADER_LAYERS_DISABLE=${DASSDL3_TEST_VULKAN_LAYERS_DISABLE}")
        endif()
    endforeach()
endif()
