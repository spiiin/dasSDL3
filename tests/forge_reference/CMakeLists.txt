cmake_minimum_required(VERSION 3.24)
project(ForgeOriginalReference LANGUAGES C CXX)
set(FORGE_SOURCE_DIR "" CACHE PATH "Unmodified forge-gpu 31f44fce checkout")
if(NOT EXISTS "${FORGE_SOURCE_DIR}/common/math/forge_math.h")
    message(FATAL_ERROR "Set FORGE_SOURCE_DIR to the original forge-gpu checkout")
endif()
find_package(SDL3 3.4.16 EXACT CONFIG REQUIRED)
set(CMAKE_C_STANDARD 99)
add_library(reference_cjson STATIC "${FORGE_SOURCE_DIR}/third_party/cJSON/cJSON.c")
foreach(lesson 10-basic-lighting 14-environment-mapping 15-cascaded-shadow-maps 16-blending 17-normal-maps 18-blinn-phong-materials 19-debug-lines 20-linear-fog)
    string(SUBSTRING "${lesson}" 0 2 number)
    add_executable(original_${number} WIN32 "${FORGE_SOURCE_DIR}/lessons/gpu/${lesson}/main.c")
    target_include_directories(original_${number} PRIVATE "${FORGE_SOURCE_DIR}/common" "${FORGE_SOURCE_DIR}/third_party/cJSON")
    target_compile_definitions(original_${number} PRIVATE FORGE_CAPTURE)
    target_link_libraries(original_${number} PRIVATE SDL3::SDL3 reference_cjson)
    set_target_properties(original_${number} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    add_custom_command(TARGET original_${number} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "$<TARGET_FILE:SDL3::SDL3>" "$<TARGET_FILE_DIR:original_${number}>")
endforeach()
foreach(model Suzanne space-shuttle CesiumMilkTruck BoxTextured)
    file(COPY "${FORGE_SOURCE_DIR}/assets/models/${model}" DESTINATION "${CMAKE_BINARY_DIR}/bin/assets/models")
endforeach()
file(COPY "${FORGE_SOURCE_DIR}/assets/skyboxes/milkyway" DESTINATION "${CMAKE_BINARY_DIR}/bin/assets/skyboxes")
foreach(lesson 16-blending 17-normal-maps)
    file(COPY "${FORGE_SOURCE_DIR}/lessons/gpu/${lesson}/assets/" DESTINATION "${CMAKE_BINARY_DIR}/bin/assets")
endforeach()
