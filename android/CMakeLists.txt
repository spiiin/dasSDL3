cmake_minimum_required(VERSION 3.24)
project(dasSDL3AndroidSmoke LANGUAGES C CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "Configure with the Android NDK toolchain")
endif()
set(DASSDL3_HOST_AOT_TOOL "" CACHE FILEPATH "Native host dasSDL3 AOT emitter")
if(NOT EXISTS "${DASSDL3_HOST_AOT_TOOL}")
    message(FATAL_ERROR "Set DASSDL3_HOST_AOT_TOOL to the host executable (see docs/android.md)")
endif()
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.." root)
set(BUILD_TESTING OFF)
add_subdirectory("${root}" dasSDL3 EXCLUDE_FROM_ALL)

add_library(main SHARED main.cpp script_host.cpp touch.cpp triangle.cpp hardware.cpp)
target_link_libraries(main PRIVATE dasSDL3)
if(DASSDL3_WITH_IMGUI)
    target_sources(main PRIVATE live.cpp keyboard.cpp "${root}/ios/imgui_input.cpp")
    target_link_libraries(main PRIVATE dasSDL3_imgui libDasModuleLiveHost)
    target_compile_definitions(main PRIVATE DASSDL3_WITH_IMGUI=1)
    foreach(pair "dasImgui/widgets;imgui" "dasLiveHost/live;modules/dasLiveHost/live")
        list(GET pair 0 moduleDir)
        list(GET pair 1 mount)
        install(DIRECTORY "${root}/third_party/daScript/modules/${moduleDir}/"
            DESTINATION "assets/app/${mount}" COMPONENT android-apk FILES_MATCHING PATTERN "*.das")
    endforeach()
    install(FILES "${root}/examples/libraries/01_imgui.das" "${root}/tests/imgui_widgets.das"
        "${root}/ios/imgui_widgets_input.das" "${root}/android/live.das"
        DESTINATION assets/app/ui COMPONENT android-apk)
    foreach(asset JetBrainsMono-Regular.ttf OFL.txt AUTHORS.txt)
        install(FILES "${root}/examples/libraries/assets/${asset}"
            DESTINATION assets/assets/libraries COMPONENT android-apk)
    endforeach()
    foreach(notice IMGUI FREETYPE_FTL MD4C)
        install(FILES "${root}/third_party/daScript/modules/dasImgui/licenses/${notice}.txt"
            DESTINATION assets/notices/ImGui COMPONENT android-apk)
    endforeach()
endif()
set(scripts android/smoke android/storage ios/input ios/camera ios/microphone ios/sensors examples/08_audio examples/lifecycle/01_square
    examples/44_gpu_render_pass examples/46_gpu_texture_uniform_bindings examples/48_gpu_native_graphics examples/49_gpu_native_compute
    examples/gpu_port_bgfx/common/file_shader_support)
foreach(feature IMAGE TTF NET MIXER SOUND)
    if(DASSDL3_WITH_${feature})
        string(TOLOWER "${feature}" library)
        target_link_libraries(main PRIVATE dasSDL3_${library})
        target_compile_definitions(main PRIVATE DASSDL3_WITH_${feature}=1)
        list(APPEND scripts "tests/${library}")
        FetchContent_GetProperties(SDL3_${library} SOURCE_DIR depSource)
        install(FILES "${depSource}/LICENSE.txt" DESTINATION "assets/notices/SDL_${library}" COMPONENT android-apk)
    endif()
endforeach()
if(DASSDL3_WITH_IMAGE)
    set(imageAssets alpha.png two_frames.gif sample.svg invalid.bin)
endif()
if(DASSDL3_WITH_TTF)
    list(APPEND scripts tests/ttf_shaping tests/ttf_gpu)
    set(fontAssets JetBrainsMono-Regular.ttf OFL.txt AUTHORS.txt Amiri-Regular.ttf Amiri-OFL.txt)
    FetchContent_GetProperties(harfbuzz_src SOURCE_DIR hbSource)
    FetchContent_GetProperties(freetype_src SOURCE_DIR ftSource)
    install(FILES "${hbSource}/COPYING" DESTINATION assets/notices/HarfBuzz COMPONENT android-apk)
    install(FILES "${ftSource}/docs/FTL.TXT" DESTINATION assets/notices/FreeType COMPONENT android-apk)
endif()
if(DASSDL3_WITH_MIXER OR DASSDL3_WITH_SOUND)
    set(audioAssets mixer-tone.wav)
endif()
foreach(asset IN LISTS imageAssets fontAssets audioAssets)
    install(FILES "${root}/examples/libraries/assets/${asset}" DESTINATION assets/assets/libraries COMPONENT android-apk)
endforeach()
file(GLOB daslib CONFIGURE_DEPENDS "${root}/third_party/daScript/daslib/*.das")
set(pending ${scripts})
set(inputs)
while(pending)
    list(POP_FRONT pending unit)
    set(input "${root}/${unit}.das")
    if(input IN_LIST inputs)
        continue()
    endif()
    list(APPEND inputs "${input}")
    file(STRINGS "${input}" imports REGEX "^require dassdl3/")
    foreach(line IN LISTS imports)
        string(REGEX MATCH "dassdl3/[A-Za-z0-9_]+" imported "${line}")
        list(APPEND pending "${imported}")
    endforeach()
endwhile()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${inputs})
include("${root}/cmake/AndroidAOT.cmake")
dassdl3_android_enum_profile(profile)
foreach(input IN LISTS inputs)
    file(RELATIVE_PATH rel "${root}" "${input}")
    string(REPLACE "/" "_" name "${rel}")
    set(aot "${CMAKE_CURRENT_BINARY_DIR}/${name}.aot.cpp")
    add_custom_command(OUTPUT "${aot}"
        COMMAND "${DASSDL3_HOST_AOT_TOOL}" "${input}" "${aot}" --sdl-enum-profile "${profile}"
        DEPENDS "${DASSDL3_HOST_AOT_TOOL}" "${profile}" ${inputs} ${daslib} VERBATIM)
    target_sources(main PRIVATE "${aot}")
endforeach()

# Stage the Java and native SDL from the same checkout. Gradle consumes this
# component so it never pulls a second SDL build from a Maven repository.
FetchContent_GetProperties(SDL3)
install(FILES "${root}/LICENSE" DESTINATION assets/notices/dasSDL3 COMPONENT android-apk)
foreach(dep SDL daScript)
    install(DIRECTORY "${root}/licenses/${dep}/" DESTINATION "assets/notices/${dep}" COMPONENT android-apk)
endforeach()
set(jni "jniLibs/${CMAKE_ANDROID_ARCH_ABI}")
install(TARGETS main SDL3-shared LIBRARY DESTINATION "${jni}" COMPONENT android-apk)
if(ANDROID_STL STREQUAL "c++_shared")
    install(FILES "${CMAKE_SYSROOT}/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}/libc++_shared.so"
        DESTINATION "${jni}" COMPONENT android-apk)
endif()
install(DIRECTORY "${sdl3_SOURCE_DIR}/android-project/app/src/main/java/"
    DESTINATION java COMPONENT android-apk FILES_MATCHING PATTERN "*.java")
install(DIRECTORY "${root}/third_party/daScript/daslib/"
    DESTINATION assets/app/daslib COMPONENT android-apk FILES_MATCHING PATTERN "*.das")
install(DIRECTORY "${root}/dassdl3/"
    DESTINATION assets/app/dassdl3 COMPONENT android-apk FILES_MATCHING PATTERN "*.das")
foreach(script IN LISTS scripts)
    get_filename_component(dir "${script}" DIRECTORY)
    install(FILES "${root}/${script}.das" DESTINATION "assets/app/${dir}" COMPONENT android-apk)
endforeach()
install(FILES "${root}/tests/macos_missing_aot.das"
    DESTINATION assets/app COMPONENT android-apk)
install(FILES "${root}/examples/assets/tone.wav" DESTINATION assets/assets COMPONENT android-apk)
foreach(shader triangle.vert triangle.frag bindings.vert bindings.frag native.comp)
    install(FILES "${root}/examples/assets/shaders/${shader}.spv"
        DESTINATION assets/assets/shaders COMPONENT android-apk)
endforeach()
